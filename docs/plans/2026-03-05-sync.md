**Goal:** Add gossip-style distributed sync so any thought-store instance can exchange thoughts with any other via full-state snapshot bundles.

**Architecture:** zstd-compressed tarballs containing `brain.db` + `assets/`. Export ships the whole DB (like `git clone`). Import opens the foreign DB read-only, finds new instances and thoughts by UUID deduplication, inserts them locally. Append-only thoughts + UUIDs = CRDT by construction.

**Tech Stack:** `github.com/klauspost/compress/zstd`, `archive/tar`, `database/sql`, existing SQLite store

// Checkpoint forces a WAL checkpoint, folding the WAL into the main DB file.

func (s *SQLiteStore) Checkpoint(ctx context.Context) error {

package main

import (
	"archive/tar"
	"context"
	"fmt"
	"io"
	"os"

	"github.com/klauspost/compress/zstd"
)

// ExportBundle creates a zstd-compressed tarball of the database.
// The tarball contains brain.db at the root (canonical name regardless of local path).
func ExportBundle(ctx context.Context, store *SQLiteStore, dbPath string, w io.Writer) error {
	// Checkpoint WAL into main DB file
	if err := store.Checkpoint(ctx); err != nil {
		return fmt.Errorf("checkpoint: %w", err)
	}

	// Open the DB file for reading
	dbFile, err := os.Open(dbPath)
	if err != nil {
		return fmt.Errorf("open db: %w", err)
	}
	defer dbFile.Close()

	dbInfo, err := dbFile.Stat()
	if err != nil {
		return fmt.Errorf("stat db: %w", err)
	}

	// Create zstd writer
	zw, err := zstd.NewWriter(w)
	if err != nil {
		return fmt.Errorf("create zstd writer: %w", err)
	}
	defer zw.Close()

	// Create tar writer
	tw := tar.NewWriter(zw)
	defer tw.Close()

	// Add brain.db (canonical name)
	if err := tw.WriteHeader(&tar.Header{
		Name: "brain.db",
		Mode: 0644,
		Size: dbInfo.Size(),
	}); err != nil {
		return fmt.Errorf("write tar header: %w", err)
	}

	if _, err := io.Copy(tw, dbFile); err != nil {
		return fmt.Errorf("write db to tar: %w", err)
	}

	// Add empty assets/ directory (reserved for future use)
	if err := tw.WriteHeader(&tar.Header{
		Name:     "assets/",
		Typeflag: tar.TypeDir,
		Mode:     0755,
	}); err != nil {
		return fmt.Errorf("write assets dir header: %w", err)
	}

	return nil
}
```

// unpackBundle decompresses a zstd tarball into the given directory.
func unpackBundle(r io.Reader, destDir string) error {
	zr, err := zstd.NewReader(r)
	if err != nil {
		return fmt.Errorf("create zstd reader: %w", err)
	}
	defer zr.Close()

	tr := tar.NewReader(zr)
	for {
		header, err := tr.Next()
		if err == io.EOF {
			break
		}
		if err != nil {
			return fmt.Errorf("read tar: %w", err)
		}

		target := filepath.Join(destDir, header.Name)

		// Prevent path traversal
		if !strings.HasPrefix(filepath.Clean(target), filepath.Clean(destDir)) {
			return fmt.Errorf("invalid tar path: %s", header.Name)
		}

		switch header.Typeflag {
		case tar.TypeDir:
			if err := os.MkdirAll(target, 0755); err != nil {
				return fmt.Errorf("mkdir %s: %w", target, err)
			}
		case tar.TypeReg:
			f, err := os.Create(target)
			if err != nil {
				return fmt.Errorf("create %s: %w", target, err)
			}
			if _, err := io.Copy(f, tr); err != nil {
				f.Close()
				return fmt.Errorf("write %s: %w", target, err)
			}
			f.Close()
		}
	}
	return nil
}
```

// InstanceUID returns the unique_id for a given instance integer ID.
func (s *SQLiteStore) InstanceUID(ctx context.Context, instanceID int) (string, error) {
	var uid string
	err := s.db.QueryRowContext(ctx,
		`SELECT unique_id FROM instances WHERE id = ?`, instanceID).Scan(&uid)
	if err != nil {
		return "", fmt.Errorf("instance uid: %w", err)
	}
	return uid, nil
}

// ImportInstance creates an instance record with a known unique_id (from a foreign DB).
// Returns the local integer ID. Idempotent: if the unique_id already exists, returns its ID.
func (s *SQLiteStore) ImportInstance(ctx context.Context, uniqueID, name, description string) (int, error) {
	var id int
	err := s.db.QueryRowContext(ctx,
		`SELECT id FROM instances WHERE unique_id = ?`, uniqueID).Scan(&id)
	if err == nil {
		return id, nil
	}

	res, err := s.db.ExecContext(ctx,
		`INSERT INTO instances (unique_id, name, description) VALUES (?, ?, ?)`,
		uniqueID, name, description)
	if err != nil {
		return 0, fmt.Errorf("import instance: %w", err)
	}
	insertedID, _ := res.LastInsertId()
	return int(insertedID), nil
}

// ImportThought inserts a thought with a known unique_id, preserving created_at.
// Skips silently if a thought with this unique_id already exists (idempotent).
// If embedding is nil, the thought is imported without an embedding (queued for re-embed).
func (s *SQLiteStore) ImportThought(ctx context.Context, uniqueID, content, source string, instanceID, modelID int, embedding []float32, createdAt string) error {
	// Check if already exists
	var exists int
	err := s.db.QueryRowContext(ctx,
		`SELECT 1 FROM thoughts WHERE unique_id = ?`, uniqueID).Scan(&exists)
	if err == nil {
		return nil // already imported
	}

	sourceID, err := s.SourceID(ctx, source)
	if err != nil {
		return err
	}

	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return fmt.Errorf("begin tx: %w", err)
	}
	defer tx.Rollback()

	res, err := tx.ExecContext(ctx,
		`INSERT INTO thoughts (unique_id, content, source_id, instance_id, created_at) VALUES (?, ?, ?, ?, ?)`,
		uniqueID, content, sourceID, instanceID, createdAt)
	if err != nil {
		return fmt.Errorf("import thought: %w", err)
	}

	if embedding != nil {
		thoughtID, _ := res.LastInsertId()
		_, err = tx.ExecContext(ctx,
			`INSERT INTO embeddings (thought_id, model_id, vector) VALUES (?, ?, ?)`,
			thoughtID, modelID, encodeVector(embedding))
		if err != nil {
			return fmt.Errorf("import embedding: %w", err)
		}
	}

	return tx.Commit()
}
```

// ImportResult summarizes what was imported from a sync bundle.
type ImportResult struct {
	NewInstances  int `json:"new_instances"`
	NewThoughts   int `json:"new_thoughts"`
	Skipped       int `json:"skipped"`
	ReembedQueued int `json:"reembed_queued"`
}

// ImportBundle decompresses a sync bundle and merges it into the local database.
// myInstanceUID is this instance's unique_id — thoughts originating from this
// instance are skipped (they came from us originally).
func ImportBundle(ctx context.Context, store *SQLiteStore, myInstanceUID string, localModelID int, r io.Reader) (*ImportResult, error) {
	// Unpack to temp dir
	tmpDir, err := os.MkdirTemp("", "sync-import-*")
	if err != nil {
		return nil, fmt.Errorf("create temp dir: %w", err)
	}
	defer os.RemoveAll(tmpDir)

	if err := unpackBundle(r, tmpDir); err != nil {
		return nil, fmt.Errorf("unpack: %w", err)
	}

	foreignDBPath := filepath.Join(tmpDir, "brain.db")
	foreignStore, err := NewSQLiteStore(foreignDBPath)
	if err != nil {
		return nil, fmt.Errorf("open foreign db: %w", err)
	}
	defer foreignStore.Close()

	result := &ImportResult{}

	// Step 1: Import new instances
	instanceMap, err := importInstances(ctx, store, foreignStore, result)
	if err != nil {
		return nil, fmt.Errorf("import instances: %w", err)
	}

	// Step 2: Get foreign active model info for embedding comparison
	foreignModelID, err := foreignStore.ActiveModelID(ctx)
	if err != nil {
		// Foreign DB might not have an active model — that's OK, just skip embeddings
		foreignModelID = 0
	}

	var foreignModelName, foreignModelVersion string
	if foreignModelID > 0 {
		foreignStore.db.QueryRowContext(ctx,
			`SELECT name, version FROM models WHERE id = ?`, foreignModelID,
		).Scan(&foreignModelName, &foreignModelVersion)
	}

	// Check if local active model matches foreign active model
	var localModelName, localModelVersion string
	store.db.QueryRowContext(ctx,
		`SELECT name, version FROM models WHERE id = ?`, localModelID,
	).Scan(&localModelName, &localModelVersion)

	modelsMatch := foreignModelName != "" &&
		foreignModelName == localModelName &&
		foreignModelVersion == localModelVersion

	// Step 3: Find candidate thoughts (not from my instance)
	rows, err := foreignStore.db.QueryContext(ctx,
		`SELECT t.unique_id, t.content, src.name, i.unique_id, t.created_at
		 FROM thoughts t
		 JOIN sources src ON src.id = t.source_id
		 JOIN instances i ON i.id = t.instance_id
		 WHERE i.unique_id != ?`, myInstanceUID)
	if err != nil {
		return nil, fmt.Errorf("query foreign thoughts: %w", err)
	}
	defer rows.Close()

	for rows.Next() {
		var thoughtUID, content, sourceName, instanceUID, createdAt string
		if err := rows.Scan(&thoughtUID, &content, &sourceName, &instanceUID, &createdAt); err != nil {
			return nil, fmt.Errorf("scan foreign thought: %w", err)
		}

		localInstanceID, ok := instanceMap[instanceUID]
		if !ok {
			continue // shouldn't happen, but be safe
		}

		if err := rows.Scan(&uid, &name, &desc); err != nil {
			return nil, fmt.Errorf("scan instance: %w", err)
		}

			`SELECT id FROM instances WHERE unique_id = ?`, uid).Scan(&existingID)

		}

	defer os.Remove(tmpFile.Name())
	defer tmpFile.Close()

	fmt.Fprintf(os.Stderr, "Creating bundle...\n")
	if err := ExportBundle(ctx, store, cfg.Database.Path, tmpFile); err != nil {
		return fmt.Errorf("export: %w", err)
	}
	tmpFile.Close()

	// Upload to peer
	peerURL := strings.TrimRight(cmd.To, "/") + "/api/sync"
	fmt.Fprintf(os.Stderr, "Pushing to %s...\n", peerURL)

	bundleFile, err := os.Open(tmpFile.Name())
	if err != nil {
		return fmt.Errorf("open bundle: %w", err)
	}
	defer bundleFile.Close()

	resp, err := http.Post(peerURL, "application/octet-stream", bundleFile)
	if err != nil {
		return fmt.Errorf("push bundle: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		body, _ := io.ReadAll(resp.Body)
		return fmt.Errorf("peer returned %d: %s", resp.StatusCode, body)
	}

	var result ImportResult
	json.NewDecoder(resp.Body).Decode(&result)

	fmt.Fprintf(os.Stderr, "Peer imported: %d new instances, %d new thoughts, %d skipped, %d queued for re-embed\n",
		result.NewInstances, result.NewThoughts, result.Skipped, result.ReembedQueued)
	return nil
}
```
