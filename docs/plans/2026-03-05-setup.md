```go
package main

import (
	"fmt"
	"os"

	"github.com/charmbracelet/huh"
	"github.com/charmbracelet/lipgloss"
)

var (
	headerStyle = lipgloss.NewStyle().
			Bold(true).
			Foreground(lipgloss.Color("12")).
			PaddingTop(1).
			PaddingBottom(1)

	successStyle = lipgloss.NewStyle().
			Foreground(lipgloss.Color("10"))

	warnStyle = lipgloss.NewStyle().
			Foreground(lipgloss.Color("11"))

	errorStyle = lipgloss.NewStyle().
			Foreground(lipgloss.Color("9"))
)

func printHeader(msg string) {
	fmt.Fprintln(os.Stderr, headerStyle.Render(msg))
}

func printSuccess(msg string) {
	fmt.Fprintln(os.Stderr, successStyle.Render("  ✓ "+msg))
}

func printWarn(msg string) {
	fmt.Fprintln(os.Stderr, warnStyle.Render("  ! "+msg))
}

func printError(msg string) {
	fmt.Fprintln(os.Stderr, errorStyle.Render("  ✗ "+msg))
}

func confirmPrompt(title string, defaultVal bool) (bool, error) {
	var result bool
	err := huh.NewConfirm().
		Title(title).
		Value(&result).
		Affirmative("Yes").
		Negative("No").
		Run()
	return result, err
}

func inputPrompt(title, defaultVal string) (string, error) {
	var result string
	err := huh.NewInput().
		Title(title).
		Value(&result).
		Placeholder(defaultVal).
		Run()
	if err != nil {
		return "", err
	}
	if result == "" {
		return defaultVal, nil
	}
	return result, nil
}

func selectPrompt(title string, options []string) (string, error) {
	var result string
	opts := make([]huh.Option[string], len(options))
	for i, o := range options {
		opts[i] = huh.NewOption(o, o)
	}
	err := huh.NewSelect[string]().
		Title(title).
		Options(opts...).
		Value(&result).
		Run()
	return result, err
}
```

```go
package main

import (
	"fmt"
	"os"
	"os/exec"
	"runtime"
)

type SetupCmd struct {
	Config string `long:"config" default:"thought-store.toml" description:"Path to config file to generate"`
}

type packageManager struct {
	name string // "paru", "yay", "pacman", "brew"
	path string // full path from LookPath
}

// packageNames maps our dependency names to package manager-specific package names.
var packageNames = map[string]map[string]string{
	"postgresql": {
		"pacman": "postgresql",
		"paru":   "postgresql",
		"yay":    "postgresql",
		"brew":   "postgresql@17",
	},
	"ollama": {
		"pacman": "ollama",
		"paru":   "ollama",
		"yay":    "ollama",
		"brew":   "ollama",
	},
}

func detectPackageManager() *packageManager {
	// Priority: paru > yay > pacman > brew
	for _, name := range []string{"paru", "yay", "pacman", "brew"} {
		if path, err := exec.LookPath(name); err == nil {
			return &packageManager{name: name, path: path}
		}
	}
	return nil
}

// installCommand returns the full command to install a package.
func (pm *packageManager) installCommand(pkg string) (string, []string) {
	pkgName := pkg
	if names, ok := packageNames[pkg]; ok {
		if n, ok := names[pm.name]; ok {
			pkgName = n
		}
	}

	switch pm.name {
	case "paru", "yay":
		return pm.path, []string{"-S", "--noconfirm", pkgName}
	case "pacman":
		return "sudo", []string{pm.path, "-S", "--noconfirm", pkgName}
	case "brew":
		return pm.path, []string{"install", pkgName}
	default:
		return "", nil
	}
}

func (cmd *SetupCmd) Execute(args []string) error {
	// Windows guard
	if runtime.GOOS == "windows" {
		// TODO: Windows service management support is planned.
		// When implemented, this will need to detect chocolatey/scoop/winget
		// and handle Windows-specific Postgres and Ollama installation.
		fmt.Fprintln(os.Stderr, "Windows setup is not yet supported.")
		fmt.Fprintln(os.Stderr, "Please install PostgreSQL (with pgvector) and Ollama manually,")
		fmt.Fprintln(os.Stderr, "then create a thought-store.toml config file. See README.md.")
		return nil
	}

	printHeader("Thought Store Setup")

	// Step 1: Config file check
	if err := cmd.checkConfigFile(); err != nil {
		return err
	}

	// Step 2: Detect package manager
	pm := detectPackageManager()
	if pm != nil {
		printSuccess(fmt.Sprintf("Found package manager: %s", pm.name))
	} else {
		printWarn("No supported package manager found (pacman/yay/paru/brew)")
		printWarn("You'll need to install dependencies manually")
	}

	cfg := &Config{
		Server: ServerConfig{Addr: ":3000"},
		Ollama: OllamaConfig{
			URL:   "http://localhost:11434",
			Model: "mxbai-embed-large",
		},
	}

	// Step 3: Postgres
	if err := cmd.setupPostgres(cfg, pm); err != nil {
		return fmt.Errorf("postgres setup: %w", err)
	}

	// Step 4: Ollama
	if err := cmd.setupOllama(cfg, pm); err != nil {
		return fmt.Errorf("ollama setup: %w", err)
	}

	// Step 5: Server config
	if err := cmd.setupServer(cfg); err != nil {
		return fmt.Errorf("server setup: %w", err)
	}

	// Step 6: mDNS (optional)
	if err := cmd.setupMDNS(cfg); err != nil {
		return fmt.Errorf("mdns setup: %w", err)
	}

	// Step 7: Write config
	if err := cmd.writeConfig(cfg); err != nil {
		return fmt.Errorf("write config: %w", err)
	}

	return nil
}

func (cmd *SetupCmd) checkConfigFile() error {
	if _, err := os.Stat(cmd.Config); err == nil {
		printWarn(fmt.Sprintf("Config file %s already exists.", cmd.Config))
		cont, err := confirmPrompt("Setup will replace it. Continue?", false)
		if err != nil {
			return err
		}
		if !cont {
			fmt.Fprintln(os.Stderr, "Setup cancelled.")
			return fmt.Errorf("cancelled")
		}
	}
	return nil
}
```

```go
func (cmd *SetupCmd) setupPostgres(cfg *Config, pm *packageManager) error {
	printHeader("PostgreSQL")

	// Ask: local or remote?
	pgLocation, err := selectPrompt(
		"Where is your PostgreSQL server?",
		[]string{"Local (this machine)", "Remote (external server)"},
	)
	if err != nil {
		return err
	}

	if pgLocation == "Remote (external server)" {
		return cmd.setupRemotePostgres(cfg)
	}

	return cmd.setupLocalPostgres(cfg, pm)
}

func (cmd *SetupCmd) setupRemotePostgres(cfg *Config) error {
	fmt.Fprintln(os.Stderr, "  Enter the full Postgres connection URL.")
	fmt.Fprintln(os.Stderr, "  Example: postgres://user:pass@host:5432/dbname?sslmode=require")

	dbURL, err := inputPrompt("Database URL", "")
	if err != nil {
		return err
	}
	if dbURL == "" {
		return fmt.Errorf("database URL is required for remote Postgres")
	}

	// Validate connection
	fmt.Fprintln(os.Stderr, "  Checking connection...")
	if err := checkPsqlConnection(dbURL); err != nil {
		printError(fmt.Sprintf("Cannot connect: %v", err))
		return fmt.Errorf("remote postgres connection failed")
	}
	printSuccess("Connected to remote PostgreSQL")

	// Check pgvector
	if err := checkPgvectorInstalled(dbURL); err != nil {
		printError("pgvector extension is not available on this server")
		fmt.Fprintln(os.Stderr, "  Ask your database administrator to install the pgvector extension.")
		return fmt.Errorf("pgvector not available on remote server")
	}
	printSuccess("pgvector extension available")

	cfg.Database.URL = dbURL
	return nil
}

func (cmd *SetupCmd) setupLocalPostgres(cfg *Config, pm *packageManager) error {
	// Check if psql is on PATH
	if _, err := exec.LookPath("psql"); err != nil {
		printWarn("PostgreSQL is not installed")
		if pm == nil {
			fmt.Fprintln(os.Stderr, "  Please install PostgreSQL manually and re-run setup.")
			return fmt.Errorf("postgresql not installed and no package manager available")
		}

		install, err := confirmPrompt(
			fmt.Sprintf("Install PostgreSQL via %s?", pm.name),
			true,
		)
		if err != nil {
			return err
		}
		if !install {
			fmt.Fprintln(os.Stderr, "  Please install PostgreSQL manually and re-run setup.")
			return fmt.Errorf("postgresql installation declined")
		}

		if err := runPackageInstall(pm, "postgresql"); err != nil {
			return fmt.Errorf("install postgresql: %w", err)
		}
		printSuccess("PostgreSQL installed")
	} else {
		printSuccess("PostgreSQL is installed")
	}

	// Check if server is running
	if !isPostgresRunning() {
		printWarn("PostgreSQL server is not running")
		fmt.Fprintln(os.Stderr, "  Start it with: sudo systemctl start postgresql")
		fmt.Fprintln(os.Stderr, "  (You may also need: sudo systemctl enable postgresql)")

		cont, err := confirmPrompt("Is PostgreSQL running now?", false)
		if err != nil {
			return err
		}
		if !cont || !isPostgresRunning() {
			return fmt.Errorf("postgresql is not running")
		}
	}
	printSuccess("PostgreSQL server is running")

	// Gather app credentials
	appUser, err := inputPrompt("App database user", "thoughtstore")
	if err != nil {
		return err
	}

	needsPassword, err := confirmPrompt("Does this user require a password?", true)
	if err != nil {
		return err
	}

	var appPassword string
	if needsPassword {
		appPassword, err = inputPrompt("Password", "thoughtstore")
		if err != nil {
			return err
		}
	}

	appDB, err := inputPrompt("Database name", "thoughtstore")
	if err != nil {
		return err
	}

	adminUser, err := inputPrompt("Admin user (for creating DB/user)", "postgres")
	if err != nil {
		return err
	}

	// Provision: create user
	fmt.Fprintln(os.Stderr, "  Provisioning database...")

	if userExists(adminUser, appUser) {
		printSuccess(fmt.Sprintf("User %q already exists", appUser))
	} else {
		if err := createDBUser(adminUser, appUser, appPassword); err != nil {
			return fmt.Errorf("create user: %w", err)
		}
		printSuccess(fmt.Sprintf("Created user %q", appUser))
	}

	// Provision: create database
	if dbExists(adminUser, appDB) {
		printSuccess(fmt.Sprintf("Database %q already exists", appDB))
	} else {
		if err := createDatabase(adminUser, appDB, appUser); err != nil {
			return fmt.Errorf("create database: %w", err)
		}
		printSuccess(fmt.Sprintf("Created database %q", appDB))
	}

	// Provision: enable pgvector
	if err := enablePgvector(adminUser, appDB); err != nil {
		printError(fmt.Sprintf("Failed to enable pgvector: %v", err))
		fmt.Fprintln(os.Stderr, "  You may need to install the pgvector package for your PostgreSQL version.")
		if pm != nil {
			fmt.Fprintln(os.Stderr, "  On Arch: sudo pacman -S postgresql-libs (pgvector may need AUR)")
		}
		return fmt.Errorf("enable pgvector: %w", err)
	}
	printSuccess("pgvector extension enabled")

	// Build connection URL
	if needsPassword {
		cfg.Database.URL = fmt.Sprintf("postgres://%s:%s@localhost/%s?sslmode=disable",
			appUser, appPassword, appDB)
	} else {
		cfg.Database.URL = fmt.Sprintf("postgres://%s@localhost/%s?sslmode=disable",
			appUser, appDB)
	}

	return nil
}

// isPostgresRunning checks if we can connect to the local Postgres server.
func isPostgresRunning() bool {
	cmd := exec.Command("psql", "-U", "postgres", "-c", "SELECT 1")
	cmd.Env = append(os.Environ(), "PGCONNECT_TIMEOUT=3")
	return cmd.Run() == nil
}

// checkPsqlConnection validates a connection URL by running a simple query.
func checkPsqlConnection(dbURL string) error {
	cmd := exec.Command("psql", dbURL, "-c", "SELECT 1")
	cmd.Env = append(os.Environ(), "PGCONNECT_TIMEOUT=5")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return fmt.Errorf("%s: %w", strings.TrimSpace(string(out)), err)
	}
	return nil
}

// checkPgvectorInstalled checks if pgvector is available on a remote server.
func checkPgvectorInstalled(dbURL string) error {
	cmd := exec.Command("psql", dbURL, "-c",
		"SELECT 1 FROM pg_available_extensions WHERE name = 'vector'")
	return cmd.Run()
}

// userExists checks if a Postgres role exists.
func userExists(adminUser, user string) bool {
	cmd := exec.Command("psql", "-U", adminUser, "-tAc",
		fmt.Sprintf("SELECT 1 FROM pg_roles WHERE rolname = '%s'", user))
	out, err := cmd.Output()
	return err == nil && strings.TrimSpace(string(out)) == "1"
}

// dbExists checks if a Postgres database exists.
func dbExists(adminUser, dbName string) bool {
	cmd := exec.Command("psql", "-U", adminUser, "-tAc",
		fmt.Sprintf("SELECT 1 FROM pg_database WHERE datname = '%s'", dbName))
	out, err := cmd.Output()
	return err == nil && strings.TrimSpace(string(out)) == "1"
}

// createDBUser creates a Postgres user, optionally with a password.
func createDBUser(adminUser, user, password string) error {
	var sql string
	if password != "" {
		sql = fmt.Sprintf("CREATE USER %s WITH PASSWORD '%s'", user, password)
	} else {
		sql = fmt.Sprintf("CREATE USER %s", user)
	}
	cmd := exec.Command("psql", "-U", adminUser, "-c", sql)
	out, err := cmd.CombinedOutput()
	if err != nil {
		return fmt.Errorf("%s: %w", strings.TrimSpace(string(out)), err)
	}
	return nil
}

// createDatabase creates a Postgres database owned by the given user.
func createDatabase(adminUser, dbName, owner string) error {
	sql := fmt.Sprintf("CREATE DATABASE %s OWNER %s", dbName, owner)
	cmd := exec.Command("psql", "-U", adminUser, "-c", sql)
	out, err := cmd.CombinedOutput()
	if err != nil {
		return fmt.Errorf("%s: %w", strings.TrimSpace(string(out)), err)
	}
	return nil
}

// enablePgvector enables the pgvector extension on the given database.
func enablePgvector(adminUser, dbName string) error {
	cmd := exec.Command("psql", "-U", adminUser, "-d", dbName, "-c",
		"CREATE EXTENSION IF NOT EXISTS vector")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return fmt.Errorf("%s: %w", strings.TrimSpace(string(out)), err)
	}
	return nil
}

// runPackageInstall runs the package manager install command, showing output.
func runPackageInstall(pm *packageManager, pkg string) error {
	bin, args := pm.installCommand(pkg)
	if bin == "" {
		return fmt.Errorf("cannot build install command for %s", pm.name)
	}

	cmdStr := bin
	for _, a := range args {
		cmdStr += " " + a
	}
	fmt.Fprintf(os.Stderr, "  Running: %s\n", cmdStr)

	c := exec.Command(bin, args...)
	c.Stdout = os.Stdout
	c.Stderr = os.Stderr
	c.Stdin = os.Stdin
	return c.Run()
}
```

```go
func (cmd *SetupCmd) setupOllama(cfg *Config, pm *packageManager) error {
	printHeader("Ollama")

	// NOTE: The embedding model is locked to mxbai-embed-large because the
	// schema hardcodes vector(1024) in store.go:Migrate(). Changing the model
	// requires changing the column dimension and re-embedding all thoughts.
	// See the "Future: Embedding Dimension Decoupling" section in
	// docs/plans/2026-03-05-setup-design.md for the migration path.
	const model = "mxbai-embed-large"

	ollamaURL := cfg.Ollama.URL

	// Check if Ollama is reachable
	if !isOllamaRunning(ollamaURL) {
		// Check if ollama binary is on PATH
		if _, err := exec.LookPath("ollama"); err != nil {
			printWarn("Ollama is not installed")
			if pm == nil {
				fmt.Fprintln(os.Stderr, "  Please install Ollama manually and re-run setup.")
				fmt.Fprintln(os.Stderr, "  See: https://ollama.ai")
				return fmt.Errorf("ollama not installed and no package manager available")
			}

			install, err := confirmPrompt(
				fmt.Sprintf("Install Ollama via %s?", pm.name),
				true,
			)
			if err != nil {
				return err
			}
			if !install {
				fmt.Fprintln(os.Stderr, "  Please install Ollama manually and re-run setup.")
				return fmt.Errorf("ollama installation declined")
			}

			if err := runPackageInstall(pm, "ollama"); err != nil {
				return fmt.Errorf("install ollama: %w", err)
			}
			printSuccess("Ollama installed")
		} else {
			printSuccess("Ollama binary found")
		}

		// Check again after install
		if !isOllamaRunning(ollamaURL) {
			printWarn("Ollama service is not running")
			fmt.Fprintln(os.Stderr, "  Start it with: systemctl start ollama")
			fmt.Fprintln(os.Stderr, "  Or in a separate terminal: ollama serve")

			cont, err := confirmPrompt("Is Ollama running now?", false)
			if err != nil {
				return err
			}
			if !cont || !isOllamaRunning(ollamaURL) {
				return fmt.Errorf("ollama is not running at %s", ollamaURL)
			}
		}
	}
	printSuccess(fmt.Sprintf("Ollama is running at %s", ollamaURL))

	// Check if model is available
	fmt.Fprintf(os.Stderr, "  Checking for model %s...\n", model)
	if isModelAvailable(ollamaURL, model) {
		printSuccess(fmt.Sprintf("Model %s is available", model))
	} else {
		printWarn(fmt.Sprintf("Model %s is not pulled", model))

		pull, err := confirmPrompt(
			fmt.Sprintf("Pull %s? (this may take a few minutes)", model),
			true,
		)
		if err != nil {
			return err
		}
		if !pull {
			fmt.Fprintf(os.Stderr, "  Run manually: ollama pull %s\n", model)
			return fmt.Errorf("model %s not available", model)
		}

		if err := pullOllamaModel(model); err != nil {
			return fmt.Errorf("pull model: %w", err)
		}
		printSuccess(fmt.Sprintf("Model %s pulled", model))
	}

	return nil
}

// isOllamaRunning checks if Ollama is reachable at the given URL.
func isOllamaRunning(baseURL string) bool {
	client := &http.Client{Timeout: 3 * time.Second}
	resp, err := client.Get(baseURL + "/api/version")
	if err != nil {
		return false
	}
	resp.Body.Close()
	return resp.StatusCode == http.StatusOK
}

// isModelAvailable checks if a model is pulled in Ollama.
func isModelAvailable(baseURL, model string) bool {
	body, _ := json.Marshal(struct {
		Name string `json:"name"`
	}{Name: model})

	client := &http.Client{Timeout: 5 * time.Second}
	resp, err := client.Post(baseURL+"/api/show", "application/json",
		bytes.NewReader(body))
	if err != nil {
		return false
	}
	resp.Body.Close()
	return resp.StatusCode == http.StatusOK
}

// pullOllamaModel runs `ollama pull` as a subprocess with output visible.
func pullOllamaModel(model string) error {
	cmd := exec.Command("ollama", "pull", model)
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	return cmd.Run()
}
```

```go
		return nil
	}

	// Default to hostname
	hostname, _ := os.Hostname()
	defaultName := hostname
	if defaultName == "" {
		defaultName = "my-brain"
	}

	name, err := inputPrompt("Instance name", defaultName)
	if err != nil {
		return err
	}
	cfg.MDNS.Name = name
	printSuccess(fmt.Sprintf("Will advertise as %q", name))
	return nil
}

func (cmd *SetupCmd) writeConfig(cfg *Config) error {
	printHeader("Writing Config")

	if err := WriteConfig(cmd.Config, cfg); err != nil {
		return err
	}

	printSuccess(fmt.Sprintf("Wrote %s", cmd.Config))

	// Summary
	fmt.Fprintln(os.Stderr)
	printHeader("Setup Complete")
	fmt.Fprintf(os.Stderr, "  Server:   %s\n", cfg.Server.Addr)
	fmt.Fprintf(os.Stderr, "  Database: %s\n", cfg.Database.URL)
	fmt.Fprintf(os.Stderr, "  Ollama:   %s (model: %s)\n", cfg.Ollama.URL, cfg.Ollama.Model)
	if cfg.MDNS.Name != "" {
		fmt.Fprintf(os.Stderr, "  mDNS:     %s\n", cfg.MDNS.Name)
	}
	fmt.Fprintln(os.Stderr)
	fmt.Fprintf(os.Stderr, "  Next: ./thought-store serve --config %s\n", cmd.Config)
	fmt.Fprintln(os.Stderr)
	return nil
}
```

```go
func TestInstallCommand(t *testing.T) {
	tests := []struct {
		pm      packageManager
		pkg     string
		wantBin string
		wantPkg string
	}{
		{
			pm:      packageManager{name: "paru", path: "/usr/bin/paru"},
			pkg:     "postgresql",
			wantBin: "/usr/bin/paru",
			wantPkg: "postgresql",
		},
		{
			pm:      packageManager{name: "pacman", path: "/usr/bin/pacman"},
			pkg:     "postgresql",
			wantBin: "sudo",
			wantPkg: "postgresql",
		},
		{
			pm:      packageManager{name: "brew", path: "/opt/homebrew/bin/brew"},
			pkg:     "postgresql",
			wantBin: "/opt/homebrew/bin/brew",
			wantPkg: "postgresql@17",
		},
		{
			pm:      packageManager{name: "yay", path: "/usr/bin/yay"},
			pkg:     "ollama",
			wantBin: "/usr/bin/yay",
			wantPkg: "ollama",
		},
	}

	for _, tt := range tests {
		t.Run(tt.pm.name+"_"+tt.pkg, func(t *testing.T) {
			bin, args := tt.pm.installCommand(tt.pkg)
			if bin != tt.wantBin {
				t.Errorf("bin = %q, want %q", bin, tt.wantBin)
			}
			// Last arg should be the package name
			if len(args) == 0 {
				t.Fatal("no args returned")
			}
			gotPkg := args[len(args)-1]
			if gotPkg != tt.wantPkg {
				t.Errorf("package = %q, want %q", gotPkg, tt.wantPkg)
			}
		})
	}
}
```
