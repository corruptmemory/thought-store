# Re-Embed Thoughts Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add a `re-embed` subcommand that re-vectorizes all thoughts using the currently configured Ollama model, with batched embedding, progress reporting, and dry-run safety.

**Architecture:** New `EmbedBatch` method on `OllamaClient` uses Ollama's `/api/embed` batch endpoint (accepts `"input": [...]`, returns `"embeddings": [[...], ...]`). New `ReEmbed` method on `Store` iterates all thoughts, sends batches to Ollama, and updates embeddings in a transaction. A `ReEmbedCmd` subcommand wires it together with `--apply` flag (dry-run by default), `--batch-size` tuning, and progress output to stderr.

**Tech Stack:** Go 1.26, Ollama `/api/embed` batch endpoint, pgx batch/transaction, existing Store + OllamaClient.

**Build rule:** Always use `./build.sh` — never invoke `go build`, `go test`, `templ generate` directly.

**Key discovery:** Ollama's `/api/embed` endpoint accepts `{"model":"...","input":["text1","text2",...]}` and returns `{"embeddings":[[...],[...],...]}`. Benchmarked at 50 embeddings in ~0.5s. This is distinct from the existing `/api/embeddings` (singular) endpoint which takes a single `"prompt"` string.

### Task 1: Add `EmbedBatch` method to `OllamaClient`

**Files:**
- Modify: `ollama.go`

**Step 1: Add batch request/response types and method**

Add after the existing `Embed` method in `ollama.go`:

```go
type embedBatchRequest struct {
	Model string   `json:"model"`
	Input []string `json:"input"`
}

type embedBatchResponse struct {
	Embeddings [][]float32 `json:"embeddings"`
}

// EmbedBatch embeds multiple texts in a single call using Ollama's /api/embed endpoint.
func (c *OllamaClient) EmbedBatch(ctx context.Context, texts []string) ([][]float32, error) {
	if len(texts) == 0 {
		return nil, nil
	}

	body, err := json.Marshal(embedBatchRequest{Model: c.model, Input: texts})
	if err != nil {
		return nil, fmt.Errorf("marshal batch embed request: %w", err)
	}

	req, err := http.NewRequestWithContext(ctx, "POST", c.baseURL+"/api/embed", bytes.NewReader(body))
	if err != nil {
		return nil, fmt.Errorf("create request: %w", err)
	}
	req.Header.Set("Content-Type", "application/json")

	resp, err := c.http.Do(req)
	if err != nil {
		return nil, fmt.Errorf("ollama batch request: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("ollama returned status %d", resp.StatusCode)
	}

	var result embedBatchResponse
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		return nil, fmt.Errorf("decode batch response: %w", err)
	}

	if len(result.Embeddings) != len(texts) {
		return nil, fmt.Errorf("expected %d embeddings, got %d", len(texts), len(result.Embeddings))
	}

	return result.Embeddings, nil
}
```

```go
// ListAllContent returns all thought IDs and content for re-embedding.
func (s *Store) ListAllContent(ctx context.Context) ([]ThoughtRow, error) {
	rows, err := s.pool.Query(ctx,
		`SELECT id, content FROM thoughts ORDER BY created_at ASC`)
	if err != nil {
		return nil, fmt.Errorf("list all content: %w", err)
	}
	defer rows.Close()

	var thoughts []ThoughtRow
	for rows.Next() {
		var t ThoughtRow
		if err := rows.Scan(&t.ID, &t.Content); err != nil {
			return nil, fmt.Errorf("scan thought row: %w", err)
		}
		thoughts = append(thoughts, t)
	}
	return thoughts, rows.Err()
}
```

**Step 3: Add `UpdateEmbeddings` method to Store**

This takes a slice of (ID, embedding) pairs and updates them in a single transaction:

```go
// UpdateEmbeddings updates the embedding column for the given thought IDs in a transaction.
func (s *Store) UpdateEmbeddings(ctx context.Context, ids []string, embeddings [][]float32) error {
	tx, err := s.pool.Begin(ctx)
	if err != nil {
		return fmt.Errorf("begin transaction: %w", err)
	}
	defer tx.Rollback(ctx)

	for i, id := range ids {
		_, err := tx.Exec(ctx,
			`UPDATE thoughts SET embedding = $1 WHERE id = $2`,
			pgvector.NewVector(embeddings[i]), id)
		if err != nil {
			return fmt.Errorf("update embedding for %s: %w", id, err)
		}
	}

	return tx.Commit(ctx)
}
```

```go
func (cmd *ReEmbedCmd) Execute(args []string) error {
	cfg, err := LoadConfig(cmd.Config)
	if err != nil {
		return err
	}

	ctx := context.Background()

	store, err := NewStore(ctx, cfg.Database.URL)
	if err != nil {
		return fmt.Errorf("connect to database: %w", err)
	}
	defer store.Close()

	ollama := NewOllamaClient(cfg.Ollama.URL, cfg.Ollama.Model)

	// Pre-flight: verify Ollama is reachable and get model info
	info, err := ollama.ShowModel(ctx)
	if err != nil {
		return fmt.Errorf("cannot reach Ollama: %w", err)
	}
	fmt.Fprintf(os.Stderr, "Model: %s (%s, %s, %dd embeddings)\n",
		cfg.Ollama.Model, info.Family, info.ParameterSize, info.EmbeddingDim)

	// Load all thoughts
	thoughts, err := store.ListAllContent(ctx)
	if err != nil {
		return fmt.Errorf("load thoughts: %w", err)
	}
	fmt.Fprintf(os.Stderr, "Thoughts to re-embed: %d\n", len(thoughts))

	if len(thoughts) == 0 {
		fmt.Fprintln(os.Stderr, "Nothing to do.")
		return nil
	}

	if !cmd.Apply {
		fmt.Fprintln(os.Stderr, "\nDry run — no changes made. Use --apply to update embeddings.")
		return nil
	}

	// Process in batches
	batchSize := cmd.BatchSize
	if batchSize <= 0 {
		batchSize = 50
	}

	total := len(thoughts)
	for i := 0; i < total; i += batchSize {
		end := i + batchSize
		if end > total {
			end = total
		}
		batch := thoughts[i:end]

		// Collect texts for this batch
		texts := make([]string, len(batch))
		ids := make([]string, len(batch))
		for j, t := range batch {
			texts[j] = t.Content
			ids[j] = t.ID
		}

		// Embed batch
		embeddings, err := ollama.EmbedBatch(ctx, texts)
		if err != nil {
			return fmt.Errorf("embed batch %d-%d: %w", i+1, end, err)
		}

		// Update database
		if err := store.UpdateEmbeddings(ctx, ids, embeddings); err != nil {
			return fmt.Errorf("update batch %d-%d: %w", i+1, end, err)
		}

		fmt.Fprintf(os.Stderr, "  re-embedded %d/%d\n", end, total)
	}

	fmt.Fprintln(os.Stderr, "Done.")
	return nil
}
```

### Task 4: Add re-embed test

**Files:**

**Step 1: Write the test**

This test inserts a thought, re-embeds it, and verifies the embedding changed (or at least that the update succeeded). Uses a real Postgres + Ollama like the other integration tests.

```go
package main

import (
	"context"
	"testing"
)

func TestReEmbed(t *testing.T) {
	ollamaAvailable(t)

	ctx := context.Background()
	store, err := NewStore(ctx, "postgres://thoughtstore:thoughtstore@localhost/thoughtstore?sslmode=disable")
	if err != nil {
		t.Skip("Postgres not available")
	}
	defer store.Close()

	if err := store.Migrate(ctx); err != nil {
		t.Fatalf("Migrate: %v", err)
	}

	ollama := NewOllamaClient("http://localhost:11434", "mxbai-embed-large")

	// Insert a test thought via the normal path
	embedding, err := ollama.Embed(ctx, "re-embed test: concurrency is not parallelism")
	if err != nil {
		t.Fatalf("Embed: %v", err)
	}
	thought, err := store.Insert(ctx, "re-embed test: concurrency is not parallelism", embedding, "test")
	if err != nil {
		t.Fatalf("Insert: %v", err)
	}

	// ListAllContent should include our thought
	rows, err := store.ListAllContent(ctx)
	if err != nil {
		t.Fatalf("ListAllContent: %v", err)
	}
	found := false
	for _, r := range rows {
		if r.ID == thought.ID {
			found = true
			break
		}
	}
	if !found {
		t.Fatal("expected to find test thought in ListAllContent")
	}

	// EmbedBatch
	embeddings, err := ollama.EmbedBatch(ctx, []string{"re-embed test: concurrency is not parallelism"})
	if err != nil {
		t.Fatalf("EmbedBatch: %v", err)
	}
	if len(embeddings) != 1 {
		t.Fatalf("expected 1 embedding, got %d", len(embeddings))
	}
	if len(embeddings[0]) != 1024 {
		t.Fatalf("expected 1024 dims, got %d", len(embeddings[0]))
	}

	// UpdateEmbeddings
	err = store.UpdateEmbeddings(ctx, []string{thought.ID}, embeddings)
	if err != nil {
		t.Fatalf("UpdateEmbeddings: %v", err)
	}

	// Verify the thought is still searchable after re-embed
	results, err := store.Search(ctx, embeddings[0], 5)
	if err != nil {
		t.Fatalf("Search after re-embed: %v", err)
	}
	found = false
	for _, r := range results {
		if r.ID == thought.ID {
			found = true
			break
		}
	}
	if !found {
		t.Fatal("expected test thought in search results after re-embed")
	}

	// Cleanup
	store.pool.Exec(ctx, "DELETE FROM thoughts WHERE id = $1", thought.ID)
}
```
