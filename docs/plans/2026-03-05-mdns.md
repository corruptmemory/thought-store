# mDNS Service Discovery Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Advertise the thought-store HTTP server via mDNS/DNS-SD on the local network and allow remote clients to auto-discover it by name.

**Architecture:** New `mdns.go` file contains advertise, discover, and scan functions using `hashicorp/mdns`. The `serve` command advertises when `[mdns].name` is configured. The `mcp --remote` command discovers before building the URL list. A new `scan` subcommand finds all instances and writes the selection to the config file.

**Tech Stack:** Go 1.26, `github.com/hashicorp/mdns`, existing go-flags CLI, BurntSushi/toml for config read/write.

**Build rule:** Always use `./build.sh` — never invoke `go build`, `go test`, `templ generate` directly.

```go
package main

import (
	"context"
	"fmt"
	"net"
	"os"
	"strconv"
	"time"

	"github.com/hashicorp/mdns"
)

// AdvertiseService starts an mDNS server advertising the given service.
// Returns the server (call Shutdown() to stop) or an error.
func AdvertiseService(name, serviceType string, port int) (*mdns.Server, error) {
	host, _ := os.Hostname()
	info := []string{"path=/api"}

	service, err := mdns.NewMDNSService(name, serviceType, "", host+".", port, nil, info)
	if err != nil {
		return nil, fmt.Errorf("create mdns service: %w", err)
	}

	server, err := mdns.NewServer(&mdns.Config{Zone: service})
	if err != nil {
		return nil, fmt.Errorf("start mdns server: %w", err)
	}

	return server, nil
}

// DiscoveredService holds info about a discovered mDNS service instance.
type DiscoveredService struct {
	Name string
	Host string
	Port int
	Addr net.IP
	Info []string
}

// URL returns the HTTP URL for this discovered service.
func (d DiscoveredService) URL() string {
	host := d.Addr.String()
	if d.Addr.To4() == nil {
		host = "[" + host + "]" // IPv6
	}
	return fmt.Sprintf("http://%s:%d", host, d.Port)
}

// DiscoverByName finds a specific named instance of a service type.
// Returns URLs for all addresses of the matching instance.
func DiscoverByName(name, serviceType string, timeout time.Duration) ([]string, error) {
	entriesCh := make(chan *mdns.ServiceEntry, 8)

	var urls []string
	done := make(chan struct{})
	go func() {
		defer close(done)
		for entry := range entriesCh {
			if entry.Name == name+"."+serviceType+".local." || entry.Instance == name {
				if entry.AddrV4 != nil {
					urls = append(urls, fmt.Sprintf("http://%s:%d", entry.AddrV4, entry.Port))
				}
				if entry.AddrV6 != nil {
					urls = append(urls, fmt.Sprintf("http://[%s]:%d", entry.AddrV6, entry.Port))
				}
			}
		}
	}()

	params := mdns.DefaultParams(serviceType)
	params.Entries = entriesCh
	params.Timeout = timeout

	err := mdns.Query(params)
	close(entriesCh)
	<-done

	if err != nil {
		return nil, fmt.Errorf("mdns query: %w", err)
	}

	return urls, nil
}

// ScanAll finds all instances of a service type on the local network.
func ScanAll(serviceType string, timeout time.Duration) ([]DiscoveredService, error) {
	entriesCh := make(chan *mdns.ServiceEntry, 16)

	var services []DiscoveredService
	done := make(chan struct{})
	go func() {
		defer close(done)
		for entry := range entriesCh {
			svc := DiscoveredService{
				Name: entry.Instance,
				Host: entry.Host,
				Port: entry.Port,
				Info: entry.InfoFields,
			}
			if entry.AddrV4 != nil {
				svc.Addr = entry.AddrV4
			} else if entry.AddrV6 != nil {
				svc.Addr = entry.AddrV6
			}
			services = append(services, svc)
		}
	}()

	params := mdns.DefaultParams(serviceType)
	params.Entries = entriesCh
	params.Timeout = timeout

	err := mdns.Query(params)
	close(entriesCh)
	<-done

	if err != nil {
		return nil, fmt.Errorf("mdns scan: %w", err)
	}

	return services, nil
}

// ParsePort extracts the port number from a listen address like ":3000" or "0.0.0.0:3000".
func ParsePort(addr string) (int, error) {
	_, portStr, err := net.SplitHostPort(addr)
	if err != nil {
		return 0, fmt.Errorf("parse address %q: %w", addr, err)
	}
	port, err := strconv.Atoi(portStr)
	if err != nil {
		return 0, fmt.Errorf("parse port %q: %w", portStr, err)
	}
	return port, nil
}
```

**Step 3: Add Execute method**

```go
func (cmd *ScanCmd) Execute(args []string) error {
	cfg, err := LoadConfig(cmd.Config)
	if err != nil {
		return err
	}

	serviceType := cmd.Service
	if serviceType == "" {
		serviceType = cfg.MDNS.ServiceType()
	}

	timeout := time.Duration(cmd.Timeout) * time.Second
	if timeout <= 0 {
		timeout = 3 * time.Second
	}

	fmt.Fprintf(os.Stderr, "Scanning for %s services (%s)...\n", serviceType, timeout)

	services, err := ScanAll(serviceType, timeout)
	if err != nil {
		return fmt.Errorf("scan: %w", err)
	}

	if len(services) == 0 {
		fmt.Fprintln(os.Stderr, "No services found.")
		return nil
	}

	fmt.Fprintf(os.Stderr, "\nFound %d service(s):\n\n", len(services))
	for i, svc := range services {
		fmt.Fprintf(os.Stderr, "  %d) %s (%s:%d)\n", i+1, svc.Name, svc.Addr, svc.Port)
	}

	fmt.Fprintf(os.Stderr, "\nSelect a service (1-%d) or 0 to cancel: ", len(services))

	var choice int
	if _, err := fmt.Scanln(&choice); err != nil {
		return fmt.Errorf("read selection: %w", err)
	}

	if choice == 0 {
		fmt.Fprintln(os.Stderr, "Cancelled.")
		return nil
	}
	if choice < 1 || choice > len(services) {
		return fmt.Errorf("invalid selection: %d", choice)
	}

	selected := services[choice-1]
	fmt.Fprintf(os.Stderr, "\nSelected: %s (%s:%d)\n", selected.Name, selected.Addr, selected.Port)

	// Update config
	cfg.MDNS.Name = selected.Name
	if cmd.Service != "" {
		cfg.MDNS.Service = cmd.Service
	}

	fmt.Fprintf(os.Stderr, "Writing to %s (existing comments will be lost)...\n", cmd.Config)

	if err := WriteConfig(cmd.Config, cfg); err != nil {
		return fmt.Errorf("write config: %w", err)
	}

	fmt.Fprintf(os.Stderr, "Done. Set [mdns] name = %q\n", selected.Name)
	return nil
}
```

	f, err := os.Create(path)

		return fmt.Errorf("create config file: %w", err)

	encoder := toml.NewEncoder(f)

}

package main

import (
	"testing"
	"time"
)

func TestMDNSAdvertiseAndDiscover(t *testing.T) {
	name := "test-brain"
	serviceType := "_thought-store-test._tcp"
	port := 9999

	// Advertise
	server, err := AdvertiseService(name, serviceType, port)
	if err != nil {
		t.Skip("mDNS not available (multicast may be disabled):", err)
	}
	defer server.Shutdown()

	// Discover by name
	urls, err := DiscoverByName(name, serviceType, 2*time.Second)
	if err != nil {
		t.Fatalf("DiscoverByName: %v", err)
	}
	if len(urls) == 0 {
		t.Fatal("expected at least one discovered URL")
	}

	// Verify URL format
	for _, u := range urls {
		if u == "" {
			t.Fatal("empty URL in discovered results")
		}
		t.Logf("Discovered: %s", u)
	}

	// ScanAll should also find it
	services, err := ScanAll(serviceType, 2*time.Second)
	if err != nil {
		t.Fatalf("ScanAll: %v", err)
	}

	found := false
	for _, svc := range services {
		if svc.Name == name && svc.Port == port {
			found = true
			break
		}
	}
	if !found {
		t.Fatalf("ScanAll did not find %q among %d services", name, len(services))
	}
}

func TestParsePort(t *testing.T) {
	tests := []struct {
		addr string
		port int
		err  bool
	}{
		{":3000", 3000, false},
		{"0.0.0.0:8080", 8080, false},
		{"localhost:443", 443, false},
		{"bad", 0, true},
	}
	for _, tt := range tests {
		port, err := ParsePort(tt.addr)
		if tt.err {
			if err == nil {
				t.Errorf("ParsePort(%q) expected error", tt.addr)
			}
			continue
		}
		if err != nil {
			t.Errorf("ParsePort(%q) unexpected error: %v", tt.addr, err)
			continue
		}
		if port != tt.port {
			t.Errorf("ParsePort(%q) = %d, want %d", tt.addr, port, tt.port)
		}
	}
}
